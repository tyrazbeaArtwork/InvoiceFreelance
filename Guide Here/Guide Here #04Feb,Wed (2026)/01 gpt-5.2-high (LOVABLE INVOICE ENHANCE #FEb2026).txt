3.1 Template store (simple + fast): LocalStorage-backed repository

Start here; later you can swap to DB without rewriting UI (Repository Pattern).

```
TypeScript// src/domain/template/template.repo.ts
export type InvoiceTemplate = {
  id: string;
  name: string;
  isDefault?: boolean;
  draftPatch: Partial<import("@/domain/invoice/invoice.types").InvoiceDraft>;
};

const KEY = "invoice.templates.v1";

export function loadTemplates(): InvoiceTemplate[] {
  const raw = localStorage.getItem(KEY);
  return raw ? (JSON.parse(raw) as InvoiceTemplate[]) : [];
}

export function saveTemplates(list: InvoiceTemplate[]) {
  localStorage.setItem(KEY, JSON.stringify(list));
}

export function getDefaultTemplate(templates: InvoiceTemplate[]) {
  return templates.find(t => t.isDefault) ?? null;
}
```


3.2 Auto-apply template when creating invoice

This is the “auto set as template” behavior.

```
TypeScript// src/domain/invoice/invoice.factory.ts
import { DEFAULT_CURRENCY } from "@/config/money";
import { getDefaultTemplate, loadTemplates } from "@/domain/template/template.repo";
import type { InvoiceDraft } from "./invoice.types";

export function createNewInvoiceDraft(): InvoiceDraft {
  const templates = loadTemplates();
  const def = getDefaultTemplate(templates);

  const base: InvoiceDraft = {
    templateId: def?.id ?? null,
    currency: DEFAULT_CURRENCY,
    invoiceNo: "",
    issueDate: new Date().toISOString().slice(0, 10),
    seller: { name: "" },
    buyer: { name: "" },
    items: [],
  };

  return def ? { ...base, ...def.draftPatch, templateId: def.id } : base;
}
```

Outcome: New invoice opens already filled with the default template (seller info, notes, etc.), but still editable.

5.2 DnD handler (dnd-kit) calling `move`

```
React// src/pages/InvoicePage/items.dnd.tsx
import { DndContext } from "@dnd-kit/core";
import { SortableContext, arrayMove } from "@dnd-kit/sortable";

export function ItemsDnD({ ids, onReorder, children }: {
  ids: string[];
  onReorder: (fromIndex: number, toIndex: number) => void;
  children: React.ReactNode;
}) {
  return (
    <DndContext
      onDragEnd={(e) => {
        const { active, over } = e;
        if (!over || active.id === over.id) return;

        const from = ids.indexOf(String(active.id));
        const to = ids.indexOf(String(over.id));
        if (from >= 0 && to >= 0) onReorder(from, to);
      }}
    >
      <SortableContext items={ids}>
        {children}
      </SortableContext>
    </DndContext>
  );
}
```

How this connects (no magic step skipped):

- Your items render with `id`
- `ids = fields.map(f => f.id)`
- `onReorder(from,to)` calls RHF `move(from,to)`
- That updates the actual form `items[]` order (the correct enterprise-safe approach)

6.1 Parsing Excel using SheetJS (xlsx)

```
TypeScript// src/features/import/excel/importInvoiceFromExcel.ts
import * as XLSX from "xlsx";
import type { InvoiceDraft } from "@/domain/invoice/invoice.types";
import { DEFAULT_CURRENCY } from "@/config/money";

export async function importInvoiceFromExcel(file: File): Promise<Partial<InvoiceDraft>> {
  const buf = await file.arrayBuffer();
  const wb = XLSX.read(buf, { type: "array" });

  // Example: you’ll confirm sheet names/columns from your file
  const info = XLSX.utils.sheet_to_json<Record<string, any>>(wb.Sheets["Invoice"] ?? wb.Sheets[wb.SheetNames[0]]);
  const items = XLSX.utils.sheet_to_json<Record<string, any>>(wb.Sheets["Items"] ?? wb.Sheets[wb.SheetNames[1]]);

  return {
    currency: DEFAULT_CURRENCY,
    buyer: { name: String(info?.[0]?.BuyerName ?? "") },
    seller: { name: String(info?.[0]?.SellerName ?? "") },
    invoiceNo: String(info?.[0]?.InvoiceNo ?? ""),
    issueDate: String(info?.[0]?.IssueDate ?? ""),
    items: items.map((r) => ({
      id: crypto.randomUUID(),
      name: String(r.Service ?? r.Name ?? ""),
      description: String(r.Description ?? ""),
      quantity: Number(r.Qty ?? 1),
      unitPrice: Number(r.UnitPrice ?? r.Price ?? 0),
    })),
  };
}
```

7.2 Parse step (simple deterministic parser first)

```
TypeScript// src/features/import/image/parseInvoiceText.ts
import type { InvoiceDraft } from "@/domain/invoice/invoice.types";
import { DEFAULT_CURRENCY } from "@/config/money";

export function parseInvoiceTextToDraft(text: string): Partial<InvoiceDraft> {
  // Start deterministic: regex for invoice no, dates, totals, etc.
  const invoiceNo = text.match(/Invoice\s*(No|#)\s*[:\-]?\s*(\S+)/i)?.[2] ?? "";
  const buyerName = text.match(/Bill(?:ed)?\s*To\s*[:\-]?\s*(.+)/i)?.[1]?.trim() ?? "";

  return {
    currency: DEFAULT_CURRENCY,
    invoiceNo,
    buyer: { name: buyerName },
  };
}
```

Enterprise-grade behavior: after import, open a review screen highlighting fields extracted with low confidence, instead of silently populating wrong values.

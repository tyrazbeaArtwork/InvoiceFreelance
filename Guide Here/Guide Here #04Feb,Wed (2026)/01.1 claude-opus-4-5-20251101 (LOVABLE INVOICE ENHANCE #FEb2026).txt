📦 Part 2: Invoice Context - The Foundation

Purpose: Centralized state management for currency (defaulting to MYR), templates, and form data. This follows the Singleton Pattern for global defaults.

```
React// src/contexts/InvoiceContext.jsx
import React, { createContext, useContext, useState, useCallback } from 'react';

const InvoiceContext = createContext(null);

// Default templates for quick selection
const DEFAULT_TEMPLATES = [
  {
    id: 'standard',
    name: 'Standard Invoice',
    isDefault: true,
    data: {
      yourCompany: {
        name: '',
        address: '',
        phone: '',
        email: '',
      },
      invoiceDetails: {
        prefix: 'INV-',
        dueDays: 30,
      },
      currency: 'MYR', // Sticky MYR as main currency
      taxRate: 6, // Malaysia SST
    }
  },
  {
    id: 'receipt',
    name: 'Simple Receipt',
    isDefault: false,
    data: {
      currency: 'MYR',
      showTax: true,
      compactView: true,
    }
  },
  {
    id: 'quotation',
    name: 'Quotation Template',
    isDefault: false,
    data: {
      currency: 'MYR',
      validityDays: 14,
      showTerms: true,
    }
  }
];

export const InvoiceProvider = ({ children }) => {
  // Core state with MYR as default
  const [currency, setCurrency] = useState('MYR');
  const [templates, setTemplates] = useState(DEFAULT_TEMPLATES);
  const [activeTemplate, setActiveTemplate] = useState(DEFAULT_TEMPLATES[0]);
  const [autoTemplateEnabled, setAutoTemplateEnabled] = useState(true);

  // Items state with drag support
  const [items, setItems] = useState([
    { id: 'item-1', name: '', description: '', quantity: 1, amount: 0 }
  ]);

  // Add new custom template
  const saveAsTemplate = useCallback((templateData, templateName) => {
    const newTemplate = {
      id: `custom-${Date.now()}`,
      name: templateName,
      isDefault: false,
      isCustom: true,
      data: { ...templateData, currency: 'MYR' }, // Always include MYR
      createdAt: new Date().toISOString(),
    };
    setTemplates(prev => [...prev, newTemplate]);
    return newTemplate;
  }, []);

  // Apply template to form
  const applyTemplate = useCallback((templateId) => {
    const template = templates.find(t => t.id === templateId);
    if (template) {
      setActiveTemplate(template);
      // Currency stays MYR unless explicitly changed
      setCurrency(template.data.currency || 'MYR');
    }
  }, [templates]);

  // Reorder items (for drag-drop)
  const reorderItems = useCallback((startIndex, endIndex) => {
    setItems(prev => {
      const result = Array.from(prev);
      const [removed] = result.splice(startIndex, 1);
      result.splice(endIndex, 0, removed);
      return result;
    });
  }, []);

  const value = {
    // Currency (sticky MYR)
    currency,
    setCurrency,
    
    // Templates
    templates,
    activeTemplate,
    autoTemplateEnabled,
    setAutoTemplateEnabled,
    saveAsTemplate,
    applyTemplate,
    
    // Items with drag support
    items,
    setItems,
    reorderItems,
  };

  return (
    <InvoiceContext.Provider value={value}>
      {children}
    </InvoiceContext.Provider>
  );
};

export const useInvoice = () => {
  const context = useContext(InvoiceContext);
  if (!context) {
    throw new Error('useInvoice must be used within InvoiceProvider');
  }
  return context;
};

export default InvoiceContext;
```

```
// Main Draggable List Container
const DraggableItemList = () => {
  const { items, setItems, reorderItems, currency } = useInvoice();

  const sensors = useSensors(
    useSensor(PointerSensor),
    useSensor(KeyboardSensor, {
      coordinateGetter: sortableKeyboardCoordinates,
    })
  );

  const handleDragEnd = (event) => {
    const { active, over } = event;

    if (active.id !== over?.id) {
      const oldIndex = items.findIndex((item) => item.id === active.id);
      const newIndex = items.findIndex((item) => item.id === over.id);
      reorderItems(oldIndex, newIndex);
    }
  };

  const handleUpdateItem = (index, field, value) => {
    setItems(prev => {
      const updated = [...prev];
      updated[index] = { ...updated[index], [field]: value };
      return updated;
    });
  };

  const handleRemoveItem = (index) => {
    if (items.length > 1) {
      setItems(prev => prev.filter((_, i) => i !== index));
    }
  };

  const handleAddItem = () => {
    setItems(prev => [
      ...prev,
      {
        id: `item-${Date.now()}`,
        name: '',
        description: '',
        quantity: 1,
        amount: 0,
      }
    ]);
  };

  // Calculate grand total
  const grandTotal = items.reduce((sum, item) => {
    return sum + (item.quantity || 0) * (item.amount || 0);
  }, 0);

  return (
    <div className="space-y-4">
      {/* Header */}
      <div className="flex items-center justify-between">
        <h3 className="font-semibold text-lg">Services & Items</h3>
        <span className="text-sm text-muted-foreground">
          Drag to reorder • Currency: {currency}
        </span>
      </div>

      {/* Draggable List */}
      <DndContext
        sensors={sensors}
        collisionDetection={closestCenter}
        onDragEnd={handleDragEnd}
      >
        <SortableContext items={items} strategy={verticalListSortingStrategy}>
          {items.map((item, index) => (
            <SortableItem
              key={item.id}
              item={item}
              index={index}
              onUpdate={handleUpdateItem}
              onRemove={handleRemoveItem}
              currency={currency}
            />
          ))}
        </SortableContext>
      </DndContext>

      {/* Add Item Button */}
      <Button
        variant="outline"
        onClick={handleAddItem}
        className="w-full border-dashed"
      >
        <Plus className="h-4 w-4 mr-2" />

      </Button>

          <p className="text-2xl font-bold text-primary">
            {formatCurrency(grandTotal, currency)}
          </p>
        </div>
      </div>

  );
};
```

📊 Part 5: Excel Import Handler

Purpose: Parse attached Excel files to auto-fill invoice data. Follows the Adapter Pattern to convert Excel format to our form structure.

```
React// src/utils/excelImportHandler.js
import * as XLSX from 'xlsx';

/**
 * Maps Excel columns to our form structure
 * Based on invoice_data_1stdev.xlsx format
 */
const COLUMN_MAPPING = {
  // Invoice Info sheet
  'Business Name': 'yourCompany.name',
  'Business Address': 'yourCompany.address',
  'Business Phone': 'yourCompany.phone',
  'Business Email': 'yourCompany.email',
  'Invoice Number': 'invoiceNumber',
  'Invoice Date': 'invoiceDate',
  'Due Date': 'dueDate',
  'Customer Name': 'billTo.name',
  'Customer Address': 'billTo.address',
  'Customer Email': 'billTo.email',
  
  // Items sheet
  'Item Name': 'name',
  'Description': 'description',
  'Quantity': 'quantity',
  'Unit Price': 'amount',
  'HSN Code': 'hsnCode',
};

/**
 * Parse Excel file and return structured form data
 */
export const parseExcelForInvoice = async (file) => {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    
    reader.onload = (e) => {
      try {
        const data = new Uint8Array(e.target.result);
        const workbook = XLSX.read(data, { type: 'array' });
        
        const result = {
          invoiceInfo: {},
          items: [],
          currency: 'MYR', // Always default to MYR
        };

        // Parse Invoice Info sheet
        if (workbook.SheetNames.includes('Invoice Info')) {
          const infoSheet = XLSX.utils.sheet_to_json(
            workbook.Sheets['Invoice Info'],
            { header: 1 }
          );
          
          // Assuming key-value pairs in two columns
          infoSheet.forEach(row => {
            if (row[0] && row[1]) {
              const mappedKey = COLUMN_MAPPING[row[0]];
              if (mappedKey) {
                setNestedValue(result.invoiceInfo, mappedKey, row[1]);
              }
            }
          });
        }

        // Parse Items sheet
        if (workbook.SheetNames.includes('Items')) {
          const itemsSheet = XLSX.utils.sheet_to_json(
            workbook.Sheets['Items']
          );
          
          result.items = itemsSheet.map((row, index) => ({
            id: `excel-item-${index}`,
            name: row['Item Name'] || row['Service'] || '',
            description: row['Description'] || '',
            quantity: parseInt(row['Quantity'] || row['Qty']) || 1,
            amount: parseFloat(row['Unit Price'] || row['Rate'] || row['Amount']) || 0,
            hsnCode: row['HSN Code'] || '',
          }));
        }

        resolve(result);
      } catch (error) {
        reject(new Error(`Failed to parse Excel: ${error.message}`));
      }
    };

    reader.onerror = () => reject(new Error('Failed to read file'));
    reader.readAsArrayBuffer(file);
  });
};

/**
 * Helper to set nested object values
 * e.g., setNestedValue(obj, 'yourCompany.name', 'Acme') 
 */
const setNestedValue = (obj, path, value) => {
  const keys = path.split('.');
  let current = obj;
  
  keys.forEach((key, index) => {
    if (index === keys.length - 1) {
      current[key] = value;
    } else {
      current[key] = current[key] || {};
      current = current[key];
    }
  });
};

/**
 * Export current form data to Excel
 */
export const exportToExcel = (formData, items, filename = 'invoice_export.xlsx') => {
  const workbook = XLSX.utils.book_new();
  
  // Invoice Info sheet
  const infoData = [
    ['Field', 'Value'],
    ['Business Name', formData.yourCompany?.name || ''],
    ['Business Address', formData.yourCompany?.address || ''],
    ['Invoice Number', formData.invoiceNumber || ''],
    ['Invoice Date', formData.invoiceDate || ''],
    ['Due Date', formData.dueDate || ''],
    ['Currency', 'MYR'], // Always MYR
    ['Customer Name', formData.billTo?.name || ''],
    ['Customer Address', formData.billTo?.address || ''],
  ];
  
  const infoSheet = XLSX.utils.aoa_to_sheet(infoData);
  XLSX.utils.book_append_sheet(workbook, infoSheet, 'Invoice Info');
  
  // Items sheet
  const itemsData = [
    ['Item Name', 'Description', 'Quantity', 'Unit Price', 'Total'],
    ...items.map(item => [
      item.name,
      item.description,
      item.quantity,
      item.amount,
      item.quantity * item.amount,
    ])
  ];
  
  const itemsSheet = XLSX.utils.aoa_to_sheet(itemsData);
  XLSX.utils.book_append_sheet(workbook, itemsSheet, 'Items');
  
  // Download
  XLSX.writeFile(workbook, filename);
};
```

```
        ],
        total: 0,
      };
      
      setExtractedData(mockExtracted);
      
      if (onDataExtracted) {
        onDataExtracted(mockExtracted);
      }
    } catch (error) {
      console.error('Failed to process image:', error);
    } finally {
      setIsProcessing(false);
    }
  };

  const clearUpload = () => {
    setUploadedImage(null);
    setImagePreview(null);
    setExtractedData(null);
    if (imagePreview) {
      URL.revokeObjectURL(imagePreview);
    }
  };

  return (
    <Card className="mb-6">
      <CardHeader className="pb-3">
        <CardTitle className="text-lg flex items-center gap-2">
          <ImageIcon className="h-5 w-5" />
          Instant Invoice from Image
        </CardTitle>
      </CardHeader>
      
      <CardContent>
        {!uploadedImage ? (
          // Dropzone
          <div
            {...getRootProps()}
            className={`
              border-2 border-dashed rounded-lg p-8 text-center cursor-pointer
              transition-all duration-200
              ${isDragActive 
                ? 'border-primary bg-primary/5' 
                : 'border-gray-300 hover:border-primary hover:bg-gray-50'
              }
            `}
          >
            <input {...getInputProps()} />
            <Upload className="h-10 w-10 mx-auto text-gray-400 mb-3" />
            <p className="text-sm text-gray-600">
              {isDragActive
                ? 'Drop your invoice/receipt image here...'
                : 'Drag & drop an invoice/receipt image, or click to select'
              }
            </p>
            <p className="text-xs text-muted-foreground mt-2">
              Supports JPG, PNG, WebP • Currency will be set to {currency}
            </p>
          </div>
        ) : (
          // Preview & Actions
          <div className="space-y-4">
            <div className="relative">
              {/* Image Preview */}
              <img
                src={imagePreview}
                alt="Uploaded invoice"
                className="w-full max-h-64 object-contain rounded-lg border"
              />
              
              {/* Clear button */}
              <Button
                variant="destructive"
                size="icon"
                onClick={clearUpload}
                className="absolute top-2 right-2"
              >
                <X className="h-4 w-4" />
              </Button>
            </div>

            {/* Action Buttons */}
            <div className="flex gap-3">
              <Button
                onClick={handleProcessImage}
                disabled={isProcessing}
                className="flex-1"
              >
                {isProcessing ? (
                  <>
                    <Loader2 className="h-4 w-4 mr-2 animate-spin" />
                    Processing...
                  </>
                ) : (
                  <>
                    <Sparkles className="h-4 w-4 mr-2" />
                    Extract Invoice Data
                  </>
                )}
              </Button>
              
              <Button variant="outline" onClick={clearUpload}>
                Cancel
              </Button>
            </div>

            {/* Extracted Data Preview */}
            {extractedData && (
              <div className="p-4 bg-green-50 rounded-lg border border-green-200">
                <div className="flex items-center gap-2 mb-2">
                  <FileText className="h-4 w-4 text-green-600" />
                  <span className="font-medium text-green-700">Data Extracted!</span>
                </div>
                <p className="text-sm text-green-600">
                  Invoice #{extractedData.invoiceNumber} • Currency: {currency}
                </p>
                <p className="text-xs text-muted-foreground mt-1">
                  Review and edit the form below to confirm details
                </p>
              </div>
            )}
          </div>
        )}
      </CardContent>
    </Card>
  );
};

export default InstantInvoiceUploader;
```

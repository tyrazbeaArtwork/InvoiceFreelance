// ═══════════════════════════════════════════════════════════════
// DRAGGABLE ITEMS - Using @dnd-kit (NOT react-beautiful-dnd!)
// Community consensus: rbd is abandoned, dnd-kit is the future
// ═══════════════════════════════════════════════════════════════

import React, { useCallback } from 'react';
import {
  DndContext,
  closestCenter,
  KeyboardSensor,
  PointerSensor,
  useSensor,
  useSensors,
  DragOverlay,
} from '@dnd-kit/core';
import {
  arrayMove,
  SortableContext,
  sortableKeyboardCoordinates,
  useSortable,
  verticalListSortingStrategy,
} from '@dnd-kit/sortable';
import { CSS } from '@dnd-kit/utilities';
import { 
  GripVertical, 
  Trash2, 
  Plus,
  AlertCircle 
} from 'lucide-react';
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { formatCurrency, parseCurrencyInput, getCurrencySymbol } from '@/utils/currency';
import { createEmptyItem } from './index';

// ─────────────────────────────────────────────────────────────
// SORTABLE ITEM ROW
// ─────────────────────────────────────────────────────────────
const SortableItem = ({ item, index, onUpdate, onRemove, currency, canRemove }) => {
  const {
    attributes,
    listeners,
    setNodeRef,
    transform,
    transition,
    isDragging,
  } = useSortable({ id: item.id });

  const style = {
    transform: CSS.Transform.toString(transform),
    transition,
    zIndex: isDragging ? 50 : 'auto',
  };

  const lineTotal = (item.quantity || 0) * (item.rate || 0);

  const handleRateChange = (value) => {
    const parsed = parseCurrencyInput(value);
    onUpdate(index, 'rate', parsed);
  };

  const handleQuantityChange = (value) => {
    const qty = parseInt(value) || 0;
    onUpdate(index, 'quantity', Math.max(0, qty));
  };

  return (
    <div
      ref={setNodeRef}
      style={style}
      className={`
        relative flex items-start gap-2 p-3 mb-2 
        bg-white border rounded-lg
        ${isDragging ? 'shadow-lg ring-2 ring-primary opacity-90' : 'shadow-sm'}
        transition-shadow duration-200
      `}
    >
      {/* Drag Handle */}
      <button
        {...attributes}
        {...listeners}
        className="mt-2 p-1 cursor-grab active:cursor-grabbing hover:bg-muted rounded touch-none"
        aria-label="Drag to reorder"
      >
        <GripVertical className="h-5 w-5 text-muted-foreground" />
      </button>

      {/* Item Number */}
      <div className="flex-shrink-0 mt-2 w-6 h-6 rounded-full bg-primary/10 flex items-center justify-center">
        <span className="text-xs font-medium text-primary">{index + 1}</span>
      </div>

      {/* Fields Grid */}
      <div className="flex-1 grid grid-cols-12 gap-2">
        {/* Item Name - 4 cols */}
        <div className="col-span-12 sm:col-span-4">
          <Input
            value={item.name}
            onChange={(e) => onUpdate(index, 'name', e.target.value)}
            placeholder="Item or service name"
            className="w-full"
          />
        </div>

        {/* Description - 3 cols */}
        <div className="col-span-12 sm:col-span-3">
          <Input
            value={item.description}
            onChange={(e) => onUpdate(index, 'description', e.target.value)}
            placeholder="Description (optional)"
            className="w-full"
          />
        </div>

        {/* Quantity - 1 col */}
        <div className="col-span-4 sm:col-span-1">
          <Input
            type="number"
            min="0"
            value={item.quantity}
            onChange={(e) => handleQuantityChange(e.target.value)}
            className="w-full text-center"
            aria-label="Quantity"
          />
        </div>

        {/* Rate - 2 cols */}
        <div className="col-span-4 sm:col-span-2">
          <div className="relative">
            <span className="absolute left-2 top-1/2 -translate-y-1/2 text-muted-foreground text-sm">
              {getCurrencySymbol(currency)}
            </span>
            <Input
              type="text"
              inputMode="decimal"
              value={item.rate || ''}
              onChange={(e) => handleRateChange(e.target.value)}
              className="w-full pl-8"
              placeholder="0.00"
              aria-label="Rate"
            />
          </div>
        </div>

        {/* Total - 2 cols */}
        <div className="col-span-4 sm:col-span-2 flex items-center justify-end">
          <span className="font-medium text-sm">
            {formatCurrency(lineTotal, currency)}
          </span>
        </div>
      </div>

      {/* Remove Button */}
      <Button
        variant="ghost"
        size="icon"
        onClick={() => onRemove(index)}
        disabled={!canRemove}
        className="mt-1 text-muted-foreground hover:text-destructive"
        aria-label="Remove item"
      >
        <Trash2 className="h-4 w-4" />
      </Button>
    </div>
  );
};

// ─────────────────────────────────────────────────────────────
// MAIN DRAGGABLE ITEMS COMPONENT
// ─────────────────────────────────────────────────────────────
const DraggableItems = ({ items, onChange, currency }) => {
  // Sensors for drag detection
  const sensors = useSensors(
    useSensor(PointerSensor, {
      activationConstraint: {
        distance: 8, // Prevent accidental drags
      },
    }),
    useSensor(KeyboardSensor, {
      coordinateGetter: sortableKeyboardCoordinates,
    })
  );

  // ─────────────────────────────────────────────────────────
  // HANDLERS
  // ─────────────────────────────────────────────────────────
  const handleDragEnd = useCallback((event) => {
    const { active, over } = event;

    if (over && active.id !== over.id) {
      const oldIndex = items.findIndex((item) => item.id === active.id);
      const newIndex = items.findIndex((item) => item.id === over.id);
      
      onChange(arrayMove(items, oldIndex, newIndex));
    }
  }, [items, onChange]);

  const handleUpdateItem = useCallback((index, field, value) => {
    const updated = [...items];
    updated[index] = { ...updated[index], [field]: value };
    onChange(updated);
  }, [items, onChange]);

  const handleRemoveItem = useCallback((index) => {
    if (items.length > 1) {
      onChange(items.filter((_, i) => i !== index));
    }
  }, [items, onChange]);

  const handleAddItem = useCallback(() => {
    onChange([...items, createEmptyItem()]);
  }, [items, onChange]);

  // ─────────────────────────────────────────────────────────
  // RENDER
  // ─────────────────────────────────────────────────────────
  return (
    <div className="space-y-3">
      {/* Header */}
      <div className="hidden sm:grid grid-cols-12 gap-2 px-10 text-xs font-medium text-muted-foreground">
        <div className="col-span-4">Item</div>
        <div className="col-span-3">Description</div>
        <div className="col-span-1 text-center">Qty</div>
        <div className="col-span-2">Rate</div>
        <div className="col-span-2 text-right">Total</div>
      </div>

      {/* Draggable List */}
      <DndContext
        sensors={sensors}
        collisionDetection={closestCenter}
        onDragEnd={handleDragEnd}
      >
        <SortableContext 
          items={items.map(i => i.id)} 
          strategy={verticalListSortingStrategy}
        >
          {items.map((item, index) => (
            <SortableItem
              key={item.id}
              item={item}
              index={index}
              onUpdate={handleUpdateItem}
              onRemove={handleRemoveItem}
              currency={currency}
              canRemove={items.length > 1}
            />
          ))}
        </SortableContext>
      </DndContext>

      {/* Empty State */}
      {items.length === 0 && (
        <div className="flex flex-col items-center justify-center p-8 border-2 border-dashed rounded-lg">
          <AlertCircle className="h-8 w-8 text-muted-foreground mb-2" />
          <p className="text-sm text-muted-foreground">No items added yet</p>
        </div>
      )}

      {/* Add Item Button */}
      <Button
        variant="outline"
        onClick={handleAddItem}
        className="w-full border-dashed hover:border-primary hover:bg-primary/5"
      >
        <Plus className="h-4 w-4 mr-2" />
        Add Item
      </Button>

      {/* Drag Hint */}
      <p className="text-xs text-center text-muted-foreground">
        💡 Drag items by the handle to reorder
      </p>
    </div>
  );
};

export default DraggableItems;

// ─────────────────────────────────────────────────────────────
// TEMPLATE STORAGE HOOKS
// ─────────────────────────────────────────────────────────────
const useTemplates = () => {
  const [templates, setTemplates] = useState([]);

  // Load from localStorage on mount
  useEffect(() => {
    try {
      const stored = localStorage.getItem(STORAGE_KEY);
      if (stored) {
        setTemplates(JSON.parse(stored));
      }
    } catch (e) {
      console.error('Failed to load templates:', e);
    }
  }, []);

  // Save to localStorage on change
  useEffect(() => {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(templates));
    } catch (e) {
      console.error('Failed to save templates:', e);
    }
  }, [templates]);

  const saveTemplate = (name, data) => {
    const newTemplate = {
      id: `tpl-${Date.now()}`,
      name,
      data,
      isDefault: templates.length === 0, // First one is default
      createdAt: new Date().toISOString(),
    };
    setTemplates(prev => [...prev, newTemplate]);
    return newTemplate;
  };

  const deleteTemplate = (id) => {
    setTemplates(prev => prev.filter(t => t.id !== id));
  };

  const setDefaultTemplate = (id) => {
    setTemplates(prev => prev.map(t => ({
      ...t,
      isDefault: t.id === id
    })));
  };

  const getDefaultTemplate = () => {
    return templates.find(t => t.isDefault) || templates[0] || null;
  };

  return {
    templates,
    saveTemplate,
    deleteTemplate,
    setDefaultTemplate,
    getDefaultTemplate,
  };
};

// ═══════════════════════════════════════════════════════════════
// EXCEL IMPORT - With community-reported edge cases handled
// ═══════════════════════════════════════════════════════════════

import * as XLSX from 'xlsx';
import { parseCurrencyInput } from './currency';

/**
 * Parse Excel file for invoice data
 * Handles: dates, currency symbols, empty rows
 */
export const parseInvoiceExcel = async (file) => {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();

    reader.onload = (e) => {
      try {
        const data = new Uint8Array(e.target.result);
        const workbook = XLSX.read(data, { 
          type: 'array',
          cellDates: true, // Community fix: parse dates properly
          cellNF: true,    // Keep number formats
        });

        const result = {
          company: {},
          customer: {},
          items: [],
          invoice: {},
        };

        // ─────────────────────────────────────────────────────
        // PARSE INVOICE INFO SHEET
        // ─────────────────────────────────────────────────────
        const infoSheetName = findSheet(workbook, ['Invoice Info', 'Info', 'Details']);
        if (infoSheetName) {
          const infoSheet = XLSX.utils.sheet_to_json(
            workbook.Sheets[infoSheetName],
            { header: 1, defval: '' }
          );

          // Parse key-value pairs
          infoSheet.forEach(row => {
            if (!row[0]) return;
            
            const key = String(row[0]).toLowerCase().trim();
            const value = row[1];

            // Map to our structure
            if (key.includes('company') || key.includes('business')) {
              result.company.name = value;
            } else if (key.includes('address') && !key.includes('customer')) {
              result.company.address = value;
            } else if (key.includes('phone') && !key.includes('customer')) {
              result.company.phone = value;
            } else if (key.includes('email') && !key.includes('customer')) {
              result.company.email = value;
            } else if (key.includes('invoice') && key.includes('number')) {
              result.invoice.number = value;
            } else if (key.includes('invoice') && key.includes('date')) {
              result.invoice.date = parseExcelDate(value);
            } else if (key.includes('due')) {
              result.invoice.dueDate = parseExcelDate(value);
            } else if (key.includes('customer') && key.includes('name')) {
              result.customer.name = value;
            }
          });
        }

        // ─────────────────────────────────────────────────────
        // PARSE ITEMS SHEET
        // ─────────────────────────────────────────────────────
        const itemsSheetName = findSheet(workbook, ['Items', 'Line Items', 'Products', 'Services']);
        if (itemsSheetName) {
          const itemsData = XLSX.utils.sheet_to_json(
            workbook.Sheets[itemsSheetName],
            { defval: '' }
          );

          result.items = itemsData
            .filter(row => {
              // Skip empty rows (community pain point)
              const hasName = row['Item'] || row['Name'] || row['Description'] || row['Service'];
              return hasName;
            })
            .map((row, index) => ({
              id: `excel-${index}-${Date.now()}`,
              name: row['Item'] || row['Name'] || row['Service'] || '',
              description: row['Description'] || row['Details'] || '',
              quantity: parseInt(row['Qty'] || row['Quantity'] || 1) || 1,
              // Community fix: strip currency symbols
              rate: parseCurrencyInput(row['Rate'] || row['Price'] || row['Unit Price'] || 0),
            }));
        }

        // Validate we got something
        if (result.items.length === 0 && !result.company.name) {
          reject(new Error('Could not find invoice data in file. Check sheet names.'));
          return;
        }

        resolve(result);

      } catch (error) {
        reject(new Error(`Parse error: ${error.message}`));
      }
    };

    reader.onerror = () => reject(new Error('Failed to read file'));
    reader.readAsArrayBuffer(file);
  });
};

/**
 * Find sheet by possible names
 */
function findSheet(workbook, possibleNames) {
  for (const name of possibleNames) {
    if (workbook.SheetNames.includes(name)) {
      return name;
    }
    // Case-insensitive search
    const found = workbook.SheetNames.find(
      s => s.toLowerCase() === name.toLowerCase()
    );
    if (found) return found;
  }
  // Return first sheet as fallback
  return workbook.SheetNames[0];
}

/**
 * Parse Excel date (handles both serial numbers and strings)
 * Community pain point: Excel dates come as numbers
 */
function parseExcelDate(value) {
  if (!value) return '';
  
  // Already a Date object
  if (value instanceof Date) {
    return value.toISOString().split('T')[0];
  }
  
  // Excel serial number (days since 1900-01-01)
  if (typeof value === 'number') {
    const date = new Date((value - 25569) * 86400 * 1000);
    return date.toISOString().split('T')[0];
  }
  
  // Try parsing as string
  try {
    const parsed = new Date(value);
    if (!isNaN(parsed.getTime())) {
      return parsed.toISOString().split('T')[0];
    }
  } catch (e) {
    // Ignore
  }
  
  return '';
}

/**
 * Export invoice to Excel
 */
export const exportToExcel = (invoiceData, filename = 'invoice.xlsx') => {
  const workbook = XLSX.utils.book_new();

  // Invoice Info sheet
  const infoData = [
    ['Field', 'Value'],
    ['Company Name', invoiceData.company?.name || ''],
    ['Company Address', invoiceData.company?.address || ''],
    ['Company Phone', invoiceData.company?.phone || ''],
    ['Company Email', invoiceData.company?.email || ''],
    ['Invoice Number', invoiceData.invoice?.number || ''],
    ['Invoice Date', invoiceData.invoice?.date || ''],
    ['Due Date', invoiceData.invoice?.dueDate || ''],
    ['Customer Name', invoiceData.customer?.name || ''],
    ['Customer Address', invoiceData.customer?.address || ''],
    ['Currency', 'MYR'],
  ];
  
  const infoSheet = XLSX.utils.aoa_to_sheet(infoData);
  XLSX.utils.book_append_sheet(workbook, infoSheet, 'Invoice Info');

  // Items sheet
  const itemsData = [
    ['Item', 'Description', 'Qty', 'Rate', 'Total'],
    ...invoiceData.items.map(item => [
      item.name,
      item.description,
      item.quantity,
      item.rate,
      item.quantity * item.rate,
    ])
  ];
  
  const itemsSheet = XLSX.utils.aoa_to_sheet(itemsData);
  XLSX.utils.book_append_sheet(workbook, itemsSheet, 'Items');

  XLSX.writeFile(workbook, filename);
};
